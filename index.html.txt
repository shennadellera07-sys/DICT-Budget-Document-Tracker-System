    <!-- Firebase SDKs -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.1.0/firebase-app.js";
        import { getFirestore, collection, addDoc, updateDoc, deleteDoc, doc, onSnapshot, signInAnonymously } from "https://www.gstatic.com/firebasejs/11.1.0/firebase-firestore.js";
        import { getAuth, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.1.0/firebase-auth.js";

        // 1. Firebase Configuration
        const firebaseConfig = JSON.parse(__firebase_config);
        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);
        const auth = getAuth(app);
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'dict-r8-ledger';

        let cloudItems = [];
        let canvas, ctx, drawing = false;
        let currentUser = null;

        // 2. Authentication Logic (CRITICAL)
        async function authenticate() {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }
            } catch (err) {
                console.error("Auth Failed:", err);
                document.getElementById('live-status').innerText = "Auth Error";
            }
        }

        onAuthStateChanged(auth, (user) => {
            currentUser = user;
            if (user) {
                document.getElementById('live-status').innerText = "Cloud Sync Active";
                document.getElementById('live-status').className = "text-[10px] font-bold text-emerald-600 uppercase tracking-[0.2em] mt-1";
                startSync();
            }
        });

        // 3. Real-time Synchronization
        function startSync() {
            if (!currentUser) return;
            const q = collection(db, 'artifacts', appId, 'public', 'data', 'ledger');
            onSnapshot(q, (snapshot) => {
                cloudItems = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
                renderLedger();
            }, (error) => {
                console.error("Snapshot Error:", error);
            });
        }

        // 4. Rendering
        function renderLedger() {
            const queryText = document.getElementById('search-bar').value.toLowerCase();
            const container = document.getElementById('ledger-container');
            
            const filtered = cloudItems.filter(l => 
                l.title?.toLowerCase().includes(queryText) || 
                l.from?.toLowerCase().includes(queryText) ||
                l.submittedBy?.toLowerCase().includes(queryText)
            ).sort((a,b) => (b.timestamp || 0) - (a.timestamp || 0));

            document.getElementById('stat-total').innerText = cloudItems.length;
            document.getElementById('stat-pending').innerText = cloudItems.filter(l => !l.dateOut).length;
            document.getElementById('stat-released').innerText = cloudItems.filter(l => l.dateOut).length;
            document.getElementById('stat-urgent').innerText = cloudItems.filter(l => l.urgent).length;

            if(filtered.length === 0) {
                container.innerHTML = '<div class="py-20 text-center text-slate-300 font-black text-[10px] tracking-widest uppercase">No data found</div>';
                return;
            }

            container.innerHTML = filtered.map(l => `
                <div class="doc-card rounded-3xl p-6 flex flex-col lg:flex-row items-center gap-8 animate-card">
                    <div class="lg:w-1/3 w-full">
                        <div class="flex items-center gap-2 mb-3">
                            <span class="text-[8px] font-black bg-blue-100 text-blue-700 px-2 py-0.5 rounded uppercase">${l.category}</span>
                            ${l.urgent ? '<span class="text-[8px] font-black text-red-500 animate-pulse bg-red-50 px-2 py-0.5 rounded">● RUSH</span>' : ''}
                        </div>
                        <h3 class="text-base font-bold text-slate-900 leading-tight mb-3">${l.title}</h3>
                        <div class="flex flex-wrap gap-2">
                             <div class="px-2 py-1 bg-slate-100 text-[8px] font-bold text-slate-500 rounded-lg">FROM: ${l.from}</div>
                             <div class="px-2 py-1 bg-blue-50 text-[8px] font-bold text-blue-600 rounded-lg">BY: ${l.submittedBy}</div>
                        </div>
                    </div>
                    
                    <div class="flex-1 w-full grid grid-cols-2 gap-6 text-xs border-x border-slate-50 px-8">
                        <div>
                            <p class="text-[8px] font-black text-slate-400 uppercase mb-1">Incoming</p>
                            <p class="font-bold text-slate-700">${l.dateIn}</p>
                            <p class="text-[9px] text-slate-500 italic mt-1">Recv: ${l.handler}</p>
                        </div>
                        <div>
                            <p class="text-[8px] font-black text-slate-400 uppercase mb-1">Released</p>
                            <p class="font-bold ${l.dateOut ? 'text-emerald-500' : 'text-slate-300'}">${l.dateOut || 'PENDING'}</p>
                            <p class="text-[9px] text-slate-500 italic mt-1">To: ${l.to || '---'}</p>
                        </div>
                    </div>

                    <div class="flex items-center gap-6">
                        ${l.signature ? `<img src="${l.signature}" class="h-10 opacity-60">` : ''}
                        <div class="flex gap-2">
                            <button onclick="window.editItem('${l.id}')" class="w-10 h-10 bg-slate-50 hover:bg-blue-600 hover:text-white rounded-xl flex items-center justify-center transition-all text-slate-400">
                                <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path d="M15.232 5.232l3.536 3.536m-2.036-5.036a2.5 2.5 0 113.536 3.536L6.5 21.036H3v-3.572L16.732 3.732z" stroke-width="2" stroke-linecap="round"/></svg>
                            </button>
                            <button onclick="window.deleteItem('${l.id}')" class="w-10 h-10 bg-slate-50 hover:bg-red-500 hover:text-white rounded-xl flex items-center justify-center transition-all text-slate-400">
                                <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" stroke-width="2" stroke-linecap="round"/></svg>
                            </button>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // 5. Form Submission
        document.getElementById('ledger-form').onsubmit = async (e) => {
            e.preventDefault();
            
            if (!currentUser) {
                alert("Cloud connection not ready. Please wait a moment.");
                return;
            }

            const submitBtn = document.getElementById('submit-btn');
            const originalText = submitBtn.innerText;
            submitBtn.innerText = "SAVING...";
            submitBtn.disabled = true;

            const editId = document.getElementById('edit-id').value;
            const signatureData = canvas.toDataURL('image/png');

            const payload = {
                title: document.getElementById('f-title').value,
                category: document.getElementById('f-category').value,
                dateIn: document.getElementById('f-date-in').value,
                from: document.getElementById('f-from').value,
                submittedBy: document.getElementById('f-submitted-by').value,
                handler: document.getElementById('f-handler').value,
                dateOut: document.getElementById('f-date-out').value || "",
                to: document.getElementById('f-to').value || "",
                urgent: document.getElementById('f-urgent').checked,
                remarks: document.getElementById('f-remarks').value || "",
                signature: signatureData,
                timestamp: Date.now()
            };

            try {
                if (editId) {
                    await updateDoc(doc(db, 'artifacts', appId, 'public', 'data', 'ledger', editId), payload);
                } else {
                    await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'ledger'), payload);
                }
                closeModal();
            } catch (err) {
                console.error("Save Error:", err);
                alert("Failed to save to cloud. Error: " + err.message);
            } finally {
                submitBtn.innerText = originalText;
                submitBtn.disabled = false;
            }
        };

        // Global functions
        window.deleteItem = async (id) => {
            if (confirm("Delete this entry from cloud?")) {
                try {
                    await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'ledger', id));
                } catch (e) { alert("Delete failed."); }
            }
        };

        window.editItem = (id) => {
            const item = cloudItems.find(i => i.id === id);
            if (!item) return;
            openModal(id);
        };

        function openModal(editId = null) {
            const form = document.getElementById('ledger-form');
            form.reset();
            document.getElementById('edit-id').value = editId || "";
            document.getElementById('modal-title').innerText = editId ? "Edit Registry Record" : "New Registry Entry";
            
            if (editId) {
                const item = cloudItems.find(i => i.id === editId);
                document.getElementById('f-title').value = item.title;
                document.getElementById('f-category').value = item.category;
                document.getElementById('f-date-in').value = item.dateIn;
                document.getElementById('f-from').value = item.from;
                document.getElementById('f-submitted-by').value = item.submittedBy;
                document.getElementById('f-handler').value = item.handler;
                document.getElementById('f-date-out').value = item.dateOut || "";
                document.getElementById('f-to').value = item.to || "";
                document.getElementById('f-urgent').checked = item.urgent;
                document.getElementById('f-remarks').value = item.remarks || "";
            } else {
                document.getElementById('f-date-in').valueAsDate = new Date();
            }

            document.getElementById('master-modal').classList.replace('hidden', 'flex');
            setTimeout(initSignature, 200);
        }

        function closeModal() {
            document.getElementById('master-modal').classList.replace('flex', 'hidden');
        }

        function initSignature() {
            canvas = document.getElementById('signature-pad');
            ctx = canvas.getContext('2d');
            const ratio = window.devicePixelRatio || 1;
            canvas.width = canvas.offsetWidth * ratio;
            canvas.height = canvas.offsetHeight * ratio;
            ctx.scale(ratio, ratio);
            ctx.lineWidth = 2;
            ctx.strokeStyle = '#0f172a';
            ctx.lineCap = 'round';

            const getPos = (e) => {
                const rect = canvas.getBoundingClientRect();
                return {
                    x: (e.touches ? e.touches[0].clientX : e.clientX) - rect.left,
                    y: (e.touches ? e.touches[0].clientY : e.clientY) - rect.top
                };
            };
            canvas.onmousedown = (e) => { drawing = true; ctx.beginPath(); const p = getPos(e); ctx.moveTo(p.x, p.y); };
            canvas.onmousemove = (e) => { if(!drawing) return; const p = getPos(e); ctx.lineTo(p.x, p.y); ctx.stroke(); };
            window.onmouseup = () => drawing = false;
        }

        // Bindings
        document.getElementById('add-btn').onclick = () => openModal();
        document.getElementById('close-modal-btn').onclick = closeModal;
        document.getElementById('cancel-btn').onclick = closeModal;
        document.getElementById('clear-sig-btn').onclick = () => ctx.clearRect(0,0,canvas.width,canvas.height);
        document.getElementById('search-bar').oninput = renderLedger;

        // Init
        authenticate();
    </script>
